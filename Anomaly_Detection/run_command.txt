1. Preprocess each turbine using data_preprocessing_pipeline.py

2. Perform feature selection on one of them.
  python scripts/feature_selection.py \
  --input_file data/processed/T06_dataset_with_labels.csv \
  --output_file data/processed/T06_dataset_with_labels_selected.csv \
  --feature_list_file data/processed/selected_features.txt \
  --summary_file data/processed/feature_selection_summary.txt \
  --std_threshold 0.01 \
  --corr_threshold 0.95 \
  --min_features 10

3. Apply the selected features to the rest of the data.
python scripts/apply_feature_list.py --input_file data/processed/T01_dataset_with_labels.csv --feature_list_file data/processed/selected_features.txt --output_file data/processed/T01_dataset_selected.csv
python scripts/apply_feature_list.py --input_file data/processed/T06_dataset_with_labels.csv --feature_list_file data/processed/selected_features.txt --output_file data/processed/T06_dataset_selected.csv
python scripts/apply_feature_list.py --input_file data/processed/T07_dataset_with_labels.csv --feature_list_file data/processed/selected_features.txt --output_file data/processed/T07_dataset_selected.csv
python scripts/apply_feature_list.py --input_file data/processed/T11_dataset_with_labels.csv --feature_list_file data/processed/selected_features.txt --output_file data/processed/T11_dataset_selected.csv

4. Create chrono splits for each of them.
python scripts/train_test_split_chrono.py --input_file data/processed/T11_dataset_selected.csv --train_out data/processed/T11_train_chrono.csv --test_out data/processed/T11_test_chrono.csv

5. Add the Turbine ID as a column.
python scripts/add_turbine_id_onehot.py --input_file data/processed/T11_train_chrono.csv --output_file data/processed/T11_train_id.csv --turbine_id T11
python scripts/add_turbine_id_onehot.py --input_file data/processed/T11_test_chrono.csv --output_file data/processed/T11_test_id.csv --turbine_id T11

6. Create the .npy files individually.

7. Merge all the .npy train files.
  python scripts/merge_multi_turbine_train.py \
  --train_files \
    data/processed/T01/train.npy \
    data/processed/T06/train.npy \
    data/processed/T07/train.npy \
    data/processed/T11/train.npy \
  --test_file data/processed/T06/test.npy \
  --test_labels_file data/processed/T06/test_labels.npy \
  --output_dir Anomaly_Detection/dataset/t06

8. Run the model.
python run.py \
  --task_name anomaly_detection \
  --is_training 1 \
  --model_id allcombined \
  --model RWKV4TS \
  --data T06 \
  --root_path ./dataset/T06 \
  --features M \
  --seq_len 50 \
  --enc_in 33 \
  --dec_in 33 \
  --c_out 33 \
  --d_model 64 \
  --n_heads 4 \
  --e_layers 1 \
  --d_ff 64 \
  --batch_size 64 \
  --train_epochs 10 \
  --patience 2 \
  --learning_rate 0.0001 \
  --anomaly_ratio 2.0 \
  --min_run_length 3 \
  --use_gpu False \
  --num_workers 0 \
  --turbine_id T06 \
  --feature_mode Avg+Std \
  --failure_window_hours 48 \
  --imputation_method time_linear_ffill_bfill
